# 在 hybrid_cae_solver_final.py 中更新 solve_step 的损伤处理逻辑
    def solve_step(self, step, disp_val):
        # 1. 物理求解 (引入残余刚度 epsilon = 1e-4)
        # 将原有的 self.D 在计算刚度时做 clip
        self.D_effective = np.clip(self.D, 0.0, 0.9999) 
        
        # 原有的 Newton 迭代 (这部分我们以后会换成 PCG)
        fy = self.solve_elasticity(disp_val)
        self.compute_strains_stresses()
        
        # 2. 损伤更新 (Mazars + 阻尼)
        delta_D, eps_eq = self.compute_damage_base()
        # 增加阻尼系数，防止损伤瞬间饱和，引发矩阵奇异
        damping = 0.5 
        self.D += delta_D * damping
        self.D = np.clip(self.D, 0.0, 0.9999) # 绝对不能等于1
        
        # 3. 结果输出
        self._visualize(step)
        return fy
